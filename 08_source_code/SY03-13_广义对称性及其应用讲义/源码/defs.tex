  \ifnum \hour=0 \advance \hour by 12 \number\hour:\else \number\hour:\fi%
     \ifnum \minute<10 0\fi%
     \number\minute%

\else \advance \hour by -12 \number\hour:%
  \ifnum \minute<10 0\fi%
  \number\minute%
  \ P.M.%
\fi%
