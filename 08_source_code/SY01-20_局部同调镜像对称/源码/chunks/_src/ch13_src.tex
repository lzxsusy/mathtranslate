\section{Eilenberg–-Zilber contraction for \v{C}ech complexes}\label{sec:Eilenberg-Zilber-constraction}


\subsection{Eilenberg--Zilber contraction for chain complexes}


In this subsection,
we follow
the presentation in \cite[\S 3]{Franz} of
the classical construction of Eilenberg--Mac Lane
\cite{EilenbergMacLane1953,EilenbergMacLane1954}
to construct the \emph{Eilenberg--Zilber maps}
for the chain complexes
associated with ordered sets and establish their relations.


For an ordered set $I$,
the chain complex
$
C_{\bullet}(I)
% ---- 原文行 4680 ----
:=
\left(\bigoplus_{p\in\mathbb{Z}}C_p(I),\partial\right)
$
associated with $I$ is defined
by
$$
C_p(I)
:=
\mathbb{Z}\left<\left\{\left.(i_0,\dots,i_p)\in I^{p+1}\right|i_0<\cdots<i_p\right\}\right>
\quad\text{for }p\ge0,
$$
$C_p(I):=0$
for $p<0$,
and
\begin{align*}
\partial:C_p(I)
&\rightarrow
C_{p-1}(I)
\qquad
(p\ge 1),
% ---- 原文行 4700 ----
\\
(i_0,\dots,i_p)
&\mapsto
\sum_{\nu=0}^p
(-1)^{\nu}
(i_0,\dots,\hat{i_{\nu}},\dots,i_p).
\end{align*}
We denote by
$
(i_{\sigma(0)},\dots,i_{\sigma(p)})
:=
\operatorname{sign}(\sigma)
(i_0,\dots,i_p)
$
for $i_0<\cdots<i_p$ in $I$\ and $\sigma\in S_{p+1}$,
and
$(i_0,\dots,i_p):=0$
if any two of $i_0,\dots,i_p$ are identical.


% ---- 原文行 4720 ----



Given two ordered sets $I$ and $J$,
we consider two natural chain complexes
$$
C_{\bullet}(I\times J)
\quad\text{and}\quad
C_{\bullet}(I)\otimes C_{\bullet}(J).
$$
The first one is the chain complex associated with $I\times J$
equipped the lexicographic order.
The second one is the usual tensor product of chain complexes
$C_{\bullet}(I)$ and $C_{\bullet}(J)$,
whose components are explicitly given by
$$
(C_{\bullet}(I)\otimes C_{\bullet}(J))_t
:=
\bigoplus_{p+r=t}
\left(C_p(I)\otimes C_r(J)\right)
% ---- 原文行 4740 ----
$$
with the boundary operator
\begin{align*}
\partial:
(C_{\bullet}(I)\otimes \partial_{\bullet}(J))_t
&\rightarrow
(C_{\bullet}(I)\otimes \partial_{\bullet}(J))_{t-1}
\\[2mm]
(i_0,\dots,i_p)\otimes(j_0,\dots,j_r)
&\mapsto
\partial((i_0,\dots,i_p))\otimes(j_0,\dots,j_r)
+
(-1)^p
(i_0,\dots,i_p)\otimes\partial((j_0,\dots,j_r)).
\end{align*}



We
claim that the two chain complexes are homotopically equivalent to each other,
% ---- 原文行 4760 ----
by constructing explicit chain maps and homotopy,
called the \emph{Eilenberg--Zilber maps}:
\begin{equation}\label{eqn:EZMaps}
\begin{tikzcd}[arrow style=tikz,>=stealth,row sep=2.5em,column sep=3.5em]
C_{\bullet}(I)\otimes C_{\bullet}(J)
  \arrow[r, shift left, "S"]
%  \arrow[loop left, <-, "H"]
&
C_{\bullet}(I\times J)
  \arrow[l, shift left, "T"]
  \arrow[loop right, <-, "\mathbb{H}"]
\end{tikzcd}
\end{equation}

\begin{defn}\label{defn:EZMaps}
(1)
The \emph{Shuffle map}
is defined by
\begin{center}
\resizebox{\textwidth}{!}{$\begin{aligned}
% ---- 原文行 4780 ----
S:
C_{\bullet}(I)\otimes C_{\bullet}(J)
&\rightarrow
C_{\bullet}(I\times J)
\\[2mm]
(i_0,\dots,i_p)\otimes(j_0,\dots,j_r)
&\mapsto
\text{
\adjustbox{scale=0.8}{
$\sum_{
0=l_0\le l_1\le\cdots\le l_{p+1}=r
}
(-1)^{l_1+\cdots+l_p}
\big(
(i_0,j_0),\dots,(i_0,j_{l_0}),
(i_1,j_{l_1}),\dots,(i_1,j_{l_2}),
\dots,
(i_p,j_{l_p}),\dots,(i_p,j_r)
\big),
$}}
% ---- 原文行 4800 ----
\end{aligned}$}
\end{center}
where exactly one subscript of $i$ or $j$ increases by $+1$ in each step.

\noindent
(2)
The \emph{Alexander–Whitney map}
is defined by
\begin{align*}
\begin{split}
T:
C_{\bullet}(I\times J)
&\rightarrow
C_{\bullet}(I)\otimes C_{\bullet}(J)
\\[0mm]
\big(\alpha_0=(i_0,j_0),\dots,\alpha_t=(i_t,j_t)\big)
&\mapsto
\sum_{p=0}^t
(i_0,\dots,i_p)\otimes(j_p,\dots,j_t).
\end{split}
% ---- 原文行 4820 ----
\end{align*}

\end{defn}

It is straightforward to check that
$S$ and $T$ are chain maps
and $S\comp T$ is the identity:
$$
S\comp \partial=\partial \comp S,
\qquad
T\comp \partial=\partial \comp T,
\qquad
S\comp T=\id_{C_{\bullet}(I)\otimes C_{\bullet}(J)}.
$$



\begin{prop}\label{prp:chain-homotopy-TS}
The chain maps
$T\bullet S$
% ---- 原文行 4840 ----
and
$\id_{C_{\bullet}(I\times J)}$
are homotopically equivalent to each other.
That is,
there is a chain homotopy
$$
\mathbb{H}:
C_{\bullet}(I\times J)
\rightarrow
C_{\bullet+1}(I\times J)
$$
between
$T\bullet S$
and
$\id_{C_{\bullet}(I\times J)}$,
satisfying
$$
\mathbb{H}\comp \partial+\partial\comp \mathbb{H}
=
T\comp S
% ---- 原文行 4860 ----
-
\id_{C_{\bullet}(I\times J)}.
$$

Moreover, it additionally satisfies
$
S\comp \mathbb{H}=0.
$
\end{prop}
