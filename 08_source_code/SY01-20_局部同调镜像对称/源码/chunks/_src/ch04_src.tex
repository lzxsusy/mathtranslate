\subsection{Group actions on $A_{\infty}$-categories and orbit categories.} In this section we define orbit categories associated to a discrete group $G$ acting on an $A_{\infty}$-category $\mathcal{A}$. This would coincide with a special case of \cite{FKQ24} when $G=\mathbb{Z}$. We assume in this section that $G$ is either a finite group or a free abelian group of finite rank.
\begin{dfn}[Group Action on $A_{\infty}$-Categories]
    An \emph{action} of $G$ on $\mathcal{A}$ assigns to each $g\in G$ a quasi-equivalence $\{g^i\}_{i\geq 0}\colon\mathcal{A}\to\mathcal{A}$ of $A_{\infty}$-categories, where \[
    g^i\colon\hom_{\mathcal{A}} (L_{i-1}, L_i)\otimes\dotsb\otimes\hom_{\mathcal{A}} (L_0, L_1)\to\hom_{\mathcal{A}} (g(L_0),g(L_i))[i-1]
    \]forms an $A_{\infty}$-endofunctor of $\mathcal{A}$. We say an $G$-action is \emph{strict} if $g^k =0$ for $k\geq 2$.
\end{dfn}
\begin{dfn}\label{defn:Ainf-orbit-category}
    Given an $A_{\infty}$-category $\mathcal{A}$ with a strict $G$-action, the \emph{orbit category} of $\mathcal{A}$ under the action $G$, written as $\mathcal{A}/G$, is a category with objects $\Obj\mathcal{A}$ and morphism spaces\[
    \hom_{\mathcal{A}/G} (L,K)\coloneqq\bigoplus_{g\in G}\hom_{\mathcal{A}} (g(L),(K)).
% ---- 原文行 820 ----
    \]
    For given tuples of objects $L_0, L_1,\dotsb ,L_d$ and chosen morphisms $a_i\in\hom_{\mathcal{A}/G} (L_i, L_{i+1})$, the $A_{\infty}$-structure on $\mathcal{A}/G$ \begin{align*}
    \mu_{\mathcal{A}/G}^d &\colon\hom_{\mathcal{A}} ((g_1\dotsb g_d)(L_{d-1}),(g_1\dotsb g_{d-1})(L_d))\otimes\dotsb\otimes\hom_{\mathcal{A}} (g_1 (L_0),L_1)\\ 
    &\to\hom_{\mathcal{A}} ((g_1\dotsb g_d)(L_0),L_d)[2-d]
    \end{align*}
    is given explicitly by 
    \begin{align*}
        \mu_{\mathcal{A}/G}^d (a_{d-1}\otimes\dotsb\otimes a_0)&=\mu_{\mathcal{A}}^d ((g_1\dotsb g_d) (a_{d-1})\otimes\dotsb\otimes g_1 (a_1)\otimes a_0),
    \end{align*}
    where $a_i\in\hom_{\mathcal{A}} (L_i,g_i (L_{i+1}))$ is $G$-homogeneous, and extended $d$-linearly.
\end{dfn}
The corresponding $dg$- and ordinary category version is also analogous, and we have
\begin{lmm}
    $\mathcal{A}/G$ is an $A_{\infty}$-category.
\end{lmm}
\begin{proof}
    We verify directly that\begin{align*}
        \sum_{k,l,j}&\mu^{j+k}_{\mathcal{A}}(g_1\dotsb g_j h_1\dotsb h_l g_{j+1}\dotsb g_{j+k-1} (a_{j+k+l})\otimes\dotsb\otimes g_1\dotsb g_j h_1\dotsb h_l (p_{j+l+1})\\ 
        &\otimes g_1\dotsb g_j\mu^l_{\mathcal{A}}(h_1\dotsb h_{l-1} (a_{j+l})\otimes\dotsb\otimes a_{j+1})\otimes g_1\dotsb g_{j-1} (a_j)\otimes\dotsb\otimes a_1)\\ 
        =&\sum_{k,l,j}\mu_{\mathcal{A}}^{j+k}(g_1\dotsb g_j h_1\dotsb h_l g_{j+1}\dotsb g_{j+k-1} (a_{j+k+l})\otimes\dotsb\otimes g_1\dotsb g_j h_1\dotsb h_l (p_{j+l+1})\\ 
% ---- 原文行 840 ----
        &\otimes\mu^l_{\mathcal{A}}\left(\mathbf{g}(h_1\dotsb h_l(a_{j+l})\otimes\dotsb\otimes\mathbf{g} h_1 (a_{j+1}))\right)\otimes g_1\dotsb g_{j-1} (a_j)\otimes\dotsb\otimes a_1)=0
    \end{align*}
    where $\mathbf{g}=g_1\dotsb g_j$, using the Maurer-Cartan equation for $\mu_{\mathcal{A}}^{\bullet}$.
\end{proof}
Note that when $\mathcal{A}$ is a dg category, i.e. $\mu_{\mathcal{A}}^d=0$ for $d\geq 3$ and $G\curvearrowright\mathcal{A}$ is a strict action, then the orbit category $\mathcal{A}/\mathbb{Z}$ is exactly the dg enhanced orbit category in the sense of Fan-Keller-Qiu \cite{FKQ24}. In this case the functor induced from $1\in\mathbb{Z}$ is naturally invertible, and therefore the natural transform $q\colon Q_{\mathbb{N}}F\to Q_{\mathbb{N}}$ in \cite{FKQ24} is an equivalence of functors, so we have nothing to localize.\par
As in \cite{FKQ24}, there is a functor $\mathcal{A}\to\mathcal{A}/G$ which is identity on the set of objects, and on morphisms given by the embedding $\hom_{\mathcal{A}} (X,Y)\hookrightarrow\hom_{\mathcal{A}/G} (X,Y)=\displaystyle\bigoplus_{g\in G}\hom_{\mathcal{A}} (X,g^0(Y))$ into the component where $g=\id$. It's not hard to check that this defines a strict $A_{\infty}$-functor $Q_G\colon\mathcal{A}\to\mathcal{A}/G$.
\begin{exmp}
Let $\Lambda$ be a group,
$\mathcal{C}$ a $\field$-linear category,
and
$\phi:\Lambda\rightarrow\Aut_{\str}(\mathcal{C})$
a strict $\Lambda$-action on $\mathcal{C}$.
It induces a strict $\Lambda$-action
$\phi:\Lambda\rightarrow\Aut_{\str}(\Ch(\mathcal{C})_{\dg})$
on the dg-category $\Ch(\mathcal{C})_{\dg}$
in a straightforward way.
\end{exmp}
\subsection{Tensor products of dg-categories.} Given $A_{\infty}$-categories $\mathcal{A}$ and $\mathcal{B}$, their tensor product $\mathcal{A}\otimes\mathcal{B}$ was recently written down by Ornaghi \cite{Ornaghi25}, which is unfortunately very complicated. But when $\mathcal{A}$ and $\mathcal{B}$ are dg categories, we have a simpler formula that can be written down, and is a special situation of Ornaghi's construction, which we will recall in the following definition.
\begin{defn}
Let $\mathcal{A}$, $\mathcal{B}$ be dg-categories,
% ---- 原文行 860 ----
$\Lambda$ an abelian group,
and
$\phi:\Lambda\rightarrow\Aut_{\str}(\mathcal{A})$,
$\psi:\Lambda\rightarrow\Aut_{\str}(\mathcal{B})$
strict $\Lambda$-actions on $\mathcal{A}$, $\mathcal{B}$,
respectively.
The \emph{twisted tensor product}
$
\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}
$
of $\mathcal{A}$ and $\mathcal{B}$
with respect to the actions $\phi$ and $\psi$ of $\Lambda$
is a dg-category
consisting of the following:

\begin{itemize}
\item
%The class of objects is given by
$
\Ob\left(\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}\right)
% ---- 原文行 880 ----
:=
\Ob(\mathcal{A})\times \Ob(\mathcal{B}).
$

\item
a dg-morphism space
%between two objects $(L,R)$ and $(L',R')$ in
%$
%\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}
%$
%given by
$$
\hom_{\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}}
\left((L,R),(L',R')\right)
:=
\displaystyle\bigoplus_{\lambda\in\Lambda}
\left(
\hom_{\mathcal{A}}(\phi_{\lambda}(L),L')
\underset{\field}{\otimes}
\hom_{\mathcal{B}}(\psi_{\lambda}(R),R')
% ---- 原文行 900 ----
\right)
$$
as a cochain complex of $\field$-vector spaces.
In particular,
the differential is given by
$$
d_{\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}}
(f\otimes g)
=
d_{\mathcal{A}}f \otimes g
+
(-1)^{|f|}
f \otimes d_{\mathcal{B}}g.
$$

\item
composition of
$\Lambda$-homogeneous
dg-morphisms
$$
% ---- 原文行 920 ----
f\otimes g
\in
\hom_{\mathcal{A}}(\phi_{\lambda}(L),L')
\underset{\field}{\otimes}
\hom_{\mathcal{B}}(\psi_{\lambda}(R),R')
\subseteq
\hom_{\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}}
\left((L,R),(L',R')\right)
$$
and
$$
f'\otimes g'
\in
\hom_{\mathcal{A}}(\phi_{\lambda'}(L'),L'')
\underset{\field}{\otimes}
\hom_{\mathcal{B}}(\psi_{\lambda'}(R'),R'')
\subseteq
\hom_{\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}}
\left((L',R'),(L'',R'')\right)
$$
% ---- 原文行 940 ----
is defined by
$$
(f\otimes g)\comp(f'\otimes g')
:=
(-1)^{|g||f'|}
(\phi_{\lambda'}(f)\comp f')
\otimes
(\psi_{\lambda'}(g)\comp g')
$$
$$
\in
\hom_{\mathcal{A}}(\phi_{\lambda\lambda'}(L),L'')
\underset{\field}{\otimes}
\hom_{\mathcal{B}}(\psi_{\lambda\lambda'}(R),R'')
\subseteq
\hom_{\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}}
\left((L,R),(L'',R'')\right),
$$
and it is extended to whole dg-morphisms by $\field$-linearity.
\end{itemize}
% ---- 原文行 960 ----
When $\Lambda=\{1\}$ and $\phi=\id$, we simply write $\mathcal{A}\otimes_1\mathcal{B} =\mathcal{A}\otimes\mathcal{B}$ and call it the \emph{tensor product} of $\mathcal{A}$ and $\mathcal{B}$.
\end{defn}
\begin{defn}
Let $\mathcal{A}$, $\mathcal{B}$, $\mathcal{A}'$, $\mathcal{B}'$ be dg-categories,
and
$\phi:\mathcal{A}\rightarrow\mathcal{A}'$,
$\psi:\mathcal{B}\rightarrow\mathcal{B}'$
be
dg-functors.
The \emph{tensor product}
$\phi\otimes\psi
:
\mathcal{A}\otimes\mathcal{B}
\rightarrow
\mathcal{A}'\otimes\mathcal{B}'
$
of $\phi$ and $\psi$
is a dg-functor
consisting of
a mapping of objects
% ---- 原文行 980 ----
$
(L,R)\mapsto (\phi(L),\psi(R))
$
and
a mapping of morphisms
$
f\otimes g \mapsto \phi(f)\otimes\psi(g).
$
\end{defn}
The following proposition is straightforward.
\begin{prp}\label{prp:twisted-tensor-equal-quotient}
Let $\mathcal{A}$, $\mathcal{B}$ be dg-categories,
and
$\phi:\Lambda\rightarrow\Aut_{\str}(\mathcal{A})$,
$\psi:\Lambda\rightarrow\Aut_{\str}(\mathcal{B})$
strict $\Lambda$-actions on $\mathcal{A}$, $\mathcal{B}$,
respectively.
Consider a strict $\Lambda$-action
$\phi\otimes \psi:\Lambda\rightarrow\Aut_{\str}(\mathcal{A}\otimes\mathcal{B})$
on $\mathcal{A}\otimes\mathcal{B}$
% ---- 原文行 1000 ----
defined by
$(\phi\otimes\psi)_{\lambda}:=\phi_{\lambda}\otimes \psi_{\lambda}
:
\mathcal{A}\otimes\mathcal{B}\rightarrow\mathcal{A}\otimes\mathcal{B}$
for each $\lambda\in\Lambda$.

Then the twisted tensor product of $\mathcal{A}$ and $\mathcal{B}$
with respect to $\phi$ and $\psi$
is isomorphic to
the orbit category of $\mathcal{A}\otimes\mathcal{B}$
with respect to $\phi\otimes\psi$:
$$
\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}
\cong
(\mathcal{A}\otimes\mathcal{B})/\Lambda.
$$
\end{prp}

\begin{proof}
The class of objects for both categories is identically $\Ob(\mathcal{A})\times \Ob(\mathcal{B})$. 
% ---- 原文行 1020 ----
For any two objects $(L,R)$ and $(L',R')$, the morphism space in the twisted tensor product is:
$$
\hom_{\mathcal{A}\underset{\Lambda}{\otimes}\mathcal{B}}((L,R),(L',R'))
=
\bigoplus_{\lambda\in\Lambda}
\left(
\hom_{\mathcal{A}}(\phi_{\lambda}(L),L')
\underset{\field}{\otimes}
\hom_{\mathcal{B}}(\psi_{\lambda}(R),R')
\right).
$$
On the other hand, by definition of the orbit category, the morphism space in $(\mathcal{A}\otimes\mathcal{B})/\Lambda$ is:
\begin{align*}
\hom_{(\mathcal{A}\otimes\mathcal{B})/\Lambda}((L,R),(L',R'))
&=
\bigoplus_{\lambda\in\Lambda}
\hom_{\mathcal{A}\otimes\mathcal{B}}((\phi\otimes\psi)_\lambda(L,R), (L',R'))
\\
&=
\bigoplus_{\lambda\in\Lambda}
% ---- 原文行 1040 ----
\hom_{\mathcal{A}\otimes\mathcal{B}}((\phi_\lambda(L),\psi_\lambda(R)), (L',R'))
\\
&=
\bigoplus_{\lambda\in\Lambda}
\left(
\hom_{\mathcal{A}}(\phi_{\lambda}(L),L')
\underset{\field}{\otimes}
\hom_{\mathcal{B}}(\psi_{\lambda}(R),R')
\right).
\end{align*}

These spaces match exactly as cochain complexes of vector spaces. 
Furthermore, the composition in the orbit category $(\mathcal{A}\otimes\mathcal{B})/\Lambda$ 
coincides with the composition rule defined in the twisted tensor product. Thus, the identity map on objects and morphisms gives an isomorphism of dg-categories.
\end{proof}
